// 1) helpers (inside match /databases/{database}/documents)
function signedIn(){ return request.auth != null; }
function isAdmin(){ return signedIn() && exists(/databases/$(database)/documents/admins/$(request.auth.uid)); }
function isRep(){ return signedIn() && exists(/databases/$(database)/documents/stateReps/$(request.auth.uid)); }
function repStates(){ return get(/databases/$(database)/documents/stateReps/$(request.auth.uid)).data.states; }
function repFor(state){ return isAdmin() || (isRep() && state in repStates()); }
function isVerifiedUser(){ return signedIn() && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.get('verified', false) == true; }

// 2) the verification request (no photo in here) — one per person
match /verifications/{uid} {
  allow create: if signedIn() && request.auth.uid == uid
                && request.resource.data.uid == uid
                && request.resource.data.status == 'pending';
  allow read:   if signedIn() && (request.auth.uid == uid || repFor(resource.data.state));
  // the person can re-send (or add the AI note) until it is approved; reps/admins decide
  allow update: if signedIn() && (
                    (request.auth.uid == uid && resource.data.status != 'approved'
                       && request.resource.data.status == 'pending')
                    || repFor(resource.data.state));
  allow delete: if isAdmin();
}

// 3) the photo of the document — private, deleted right after the decision
match /verificationImages/{uid} {
  allow create, update: if signedIn() && request.auth.uid == uid && request.resource.data.uid == uid;
  allow read:   if signedIn() && (request.auth.uid == uid || repFor(resource.data.state));
  allow delete: if signedIn() && (request.auth.uid == uid || repFor(resource.data.state));
}

// 4) who the state reps are — only admins can change this list
match /stateReps/{uid} {
  allow read:  if signedIn() && (request.auth.uid == uid || isAdmin());
  allow write: if isAdmin();
}
